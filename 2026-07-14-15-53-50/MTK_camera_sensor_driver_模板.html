static kal_uint16 xxx_preview_setting[] = {
    // 例：切页 + 设输出格式 + 设分辨率 + stream on
    0x0100, 0x00,             // (reg, val)  standby
    0x0305, 0xxx,             // 分频/PLL
    0x0340, 0xxxxx,           // frame length
    ...                       // 由模组厂提供
    0x0100, 0x01,             // stream on
};

static kal_uint16 xxx_capture_setting[] = { ... };
static kal_uint16 xxx_video_setting[]   = { ... };

// 通用写表（MIPI CCI / I2C）
static void xxx_write_setting(kal_uint16 *tbl, kal_uint32 sz)
{
    for (kal_uint32 i = 0; i < sz; i += 2)
        writeReg(tbl[i], tbl[i+1]);   // 实际走 imgsensor_i2c 封装
}

4. xxx_open（第 10 份已讲，这里带全）

static kal_uint32 xxx_open(void)
{
    kal_uint16 chip_id = 0;
    readReg(XXX_CHIP_ID_REG, &chip_id);          // 如 0x300A/0x300B
    if (chip_id != XXX_SENSOR_ID)
        return ERROR_SENSOR_CONNECT_FAIL;        // ★ 配对硬件依据
    // 上电时序 / MCLK 已由 DTS + kd_camera_hw 完成
    xxx_write_setting(xxx_preview_setting,
                     sizeof(xxx_preview_setting)/sizeof(kal_uint16));
    return ERROR_NONE;
}

5. xxx_get_info（★ 本文重点）

把"这个 sensor 能跑什么分辨率/帧率/时钟/极性"告诉 framework。不同 scenarioId 填不同分辨率。

static kal_uint32 xxx_get_info(MSDK_SCENARIO_ID_ENUM scenarioId,
                              ACDK_SENSOR_INFO_STRUCT  *pSensorInfo,
                              ACDK_SENSOR_CONFIG_STRUCT *pSensorConfigData)
{
    switch (scenarioId) {
    case MSDK_SCENARIO_ID_CAMERA_PREVIEW:
        pSensorInfo->SensorPreviewResolutionX = XXX_PREVIEW_WIDTH;
        pSensorInfo->SensorPreviewResolutionY = XXX_PREVIEW_HEIGHT;
        break;
    case MSDK_SCENARIO_ID_CAMERA_CAPTURE_JPEG:
        pSensorInfo->SensorCaptureResolutionX = XXX_CAPTURE_WIDTH;
        pSensorInfo->SensorCaptureResolutionY = XXX_CAPTURE_HEIGHT;
        break;
    case MSDK_SCENARIO_ID_VIDEO_PREVIEW:
    case MSDK_SCENARIO_ID_HIGH_SPEED_VIDEO:
        pSensorInfo->SensorVideoResolutionX = XXX_VIDEO_WIDTH;
        pSensorInfo->SensorVideoResolutionY = XXX_VIDEO_HEIGHT;
        break;
    default:
        break;
    }

    // ---- 公共配置（所有 scenario 共用） ----
    pSensorInfo->SensorClockFreq         = 24;     // MCLK = 24MHz
    pSensorInfo->SensorClockPolarity     = SENSOR_CLOCK_POLARITY_LOW;
    pSensorInfo->SensorClockFallingPolarity = SENSOR_CLOCK_POLARITY_LOW;
    pSensorInfo->SensorHsyncPolarity     = SENSOR_CLOCK_POLARITY_LOW;
    pSensorInfo->SensorVsyncPolarity     = SENSOR_CLOCK_POLARITY_LOW;
    pSensorInfo->SensorInterruptDelayLines = 4;
    pSensorInfo->SensorResetActiveHigh   = FALSE;
    pSensorInfo->SensorOutputDataFormat  = SENSOR_OUTPUT_FORMAT_RAW_B;  // Bayer 排列
    pSensorInfo->SensorPreviewFrameRate  = XXX_PREVIEW_FPS;
    pSensorInfo->SensorCaptureFrameRate  = XXX_CAPTURE_FPS;
    pSensorInfo->SensorVideoFrameRate    = XXX_VIDEO_FPS;
    // MIPI 通道/速率（部分 BSP 用 SensorPreviewDataRate 等字段）
    pSensorInfo->SensorDataBitWidth      = 10;     // 10/12-bit RAW
    return ERROR_NONE;
}

ACDK_SENSOR_INFO_STRUCT 字段逐项

字段含义填法
SensorPreviewResolutionX/Y预览分辨率datasheet 有效像素，preview 档
SensorCaptureResolutionX/Y拍照分辨率capture 档（可高于预览，binning 关）
SensorVideoResolutionX/Y录像分辨率video 档
SensorClockFreqMCLK 频率(MHz)板级 MCLK，典型 24
SensorClockPolarity 等时钟/同步极性一般 LOW；按模组规格
SensorResetActiveHighreset 有效电平多数 sensor 低有效 → FALSE
SensorOutputDataFormat输出格式SENSOR_OUTPUT_FORMAT_RAW_B / _RGB565 / _YUV422
SensorPreview/Capture/VideoFrameRate各模式帧率30/60 等，与 setting 表匹配
SensorDataBitWidthRAW 位宽10 / 12
SensorInterruptDelayLinesVSYNC 中断延迟行一般 4~16，按 ISP 要求

6. xxx_control（★ 本文重点）

根据 scenarioId 切换 sensor 工作模式（下发对应寄存器序列表）。

static kal_uint32 xxx_control(MSDK_SCENARIO_ID_ENUM scenarioId,
                              MSDK_SENSOR_EXPOSURE_WINDOW_STRUCT *pImageWindow,
                              MSDK_SENSOR_CONFIG_STRUCT *pSensorConfigData)
{
    switch (scenarioId) {
    case MSDK_SCENARIO_ID_CAMERA_PREVIEW:
        xxx_write_setting(xxx_preview_setting,
                          sizeof(xxx_preview_setting)/sizeof(kal_uint16));
        break;
    case MSDK_SCENARIO_ID_CAMERA_CAPTURE_JPEG:
        xxx_write_setting(xxx_capture_setting,
                          sizeof(xxx_capture_setting)/sizeof(kal_uint16));
        break;
    case MSDK_SCENARIO_ID_VIDEO_PREVIEW:
    case MSDK_SCENARIO_ID_HIGH_SPEED_VIDEO:
        xxx_write_setting(xxx_video_setting,
                          sizeof(xxx_video_setting)/sizeof(kal_uint16));
        break;
    default:
        return ERROR_INVALID_SCENARIO_ID;
    }
    return ERROR_NONE;
}
坑：get_info 报的分辨率必须和 control 下发的 setting 表一致；否则 framework 按 X 尺寸送帧、sensor 实际输出 Y 尺寸 → 花屏/裁切。改分辨率时两处一起改。

7. xxx_feature_control（3A 能力协商）

framework 通过它问 sensor 支持哪些 feature（分辨率表、最大帧率、曝光范围等）。

static kal_uint32 xxx_feature_control(MSDK_SENSOR_FEATURE_ENUM featureId,
                                      MUINT8 *pFeaturePara)
{
    switch (featureId) {
    case SENSOR_FEATURE_GET_RESOLUTION: {
        SENSOR_RESOLUTION_INFO_STRUCT *p =
            (SENSOR_RESOLUTION_INFO_STRUCT *)pFeaturePara;
        p->SensorFullWidth   = XXX_CAPTURE_WIDTH;
        p->SensorFullHeight  = XXX_CAPTURE_HEIGHT;
        p->SensorPreviewWidth  = XXX_PREVIEW_WIDTH;
        p->SensorPreviewHeight = XXX_PREVIEW_HEIGHT;
        break;
    }
    case SENSOR_FEATURE_GET_PERIOD:
    case SENSOR_FEATURE_GET_PIXEL_CLOCK_FREQ:
        // 返回 period / pixel clock（用于帧率计算）
        break;
    case SENSOR_FEATURE_SET_ESHUTTER:
        // 写曝光行（来自 3A tuning）
        break;
    case SENSOR_FEATURE_SET_GAIN:
        // 写模拟/数字增益
        break;
    default:
        break;
    }
    return ERROR_NONE;
}

9. 排障（驱动侧）

现象查
probe 通过但画面花get_info 分辨率 ≠ control 下发的 setting 表尺寸
