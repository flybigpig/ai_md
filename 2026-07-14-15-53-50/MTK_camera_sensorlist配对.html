Camera: sensorlist.cpp ↔ kd_imgsensor 配对流程

代码级 · 核心：两端靠同一个 XXX_SENSOR_ID 配对，kernel 实测 I2C chip ID 确认硬件在板 · 原文框架由用户编写，WorkBuddy 渲染图示
MT6893Android 14imgsensorHAL↔Kernel
一句话：HAL 列表给"软件函数表"，kernel 列表做"硬件 I2C 探测 + 驱动分发"。两边用同一个 XXX_SENSOR_ID 对齐；真正"有没有这块 sensor"由 kernel xxx_open() 里读 I2C chip ID 决定。本文是第 8 份"场景三：调 camera"的底层机制展开。

二、结构体定义（两端都要一致）

HAL 侧（kd_imgsensor.h / imgsensor_sensor_list.cpp）：

C
typedef struct {
    MUINT32 sensorId;                              // ★ 配对键
    char   *driverName;
    MUINT32 (*Init)(PSENSOR_FUNCTION_STRUCT *pfFunc);
} MSDK_SENSOR_INIT_FUNCTION_STRUCT;

Kernel 侧（kernel-5.15/drivers/misc/mediatek/imgsensor/kd_imgsensor.h）：

C
typedef struct {
    UINT32 sensorId;                              // ★ 配对键(必须 == HAL 的)
    char  *driverName;
    UINT32 (*Init)(PSENSOR_FUNCTION_STRUCT *pfFunc);
    struct SENSOR_FUNCTION_STRUCT *pSensorList;   // kernel 实际函数表
} KD_IMGSENSOR_INIT_FUNCTION_STRUCT;

函数表（sensor 驱动实现，两端 Init 都返回它）：

C
typedef struct {
    UINT32 (*SensorOpen)(void);
    UINT32 (*SensorGetInfo)(...);
    UINT32 (*SensorGetResolution)(...);
    UINT32 (*SensorFeatureControl)(MUINT32 *pFeatureCtl);
    UINT32 (*SensorControl)(UINT32 FeatureId, MUINT8 *pFeaturePara);
    UINT32 (*SensorClose)(void);
} SENSOR_FUNCTION_STRUCT;

三、HAL 列表注册（imgsensor_sensor_list.cpp）

C
// vendor/mediatek/proprietary/custom/mt6893/hal/imgsensor/imgsensor_sensor_list.cpp
#include "xxx_sensor.h"          // 里面定义 XXX_SENSOR_ID / SENSOR_DRVNAME_XXX

MSDK_SENSOR_INIT_FUNCTION_STRUCT sensorList[] = {
#if defined(XXX_MIPI_RAW)
    { XXX_SENSOR_ID, SENSOR_DRVNAME_XXX, xxx_SensorInit },   // ★ ID 与 kernel 必须相同
#endif
#if defined(YYY_MIPI_RAW)
    { YYY_SENSOR_ID, SENSOR_DRVNAME_YYY, yyy_SensorInit },
#endif
    { 0, NULL, NULL }            // 哨兵结束项
};
坑：XXX_SENSOR_ID（如 0x1234）通常定义在 xxx_sensor.h，HAL 和 kernel 必须 include 同一个值。公司里常见错误就是两份头没同步改，导致 HAL 起不来。

四、Kernel 列表注册（kd_imgsensor.c）

C
// kernel-5.15/drivers/misc/mediatek/imgsensor/kd_imgsensor.c
#include "xxx_sensor.h"

KD_IMGSENSOR_INIT_FUNCTION_STRUCT g_pstSensorList[MAX_NUM_OF_SUPPORT_SENSOR] = {
#if defined(XXX_MIPI_RAW)
    { XXX_SENSOR_ID, SENSOR_DRVNAME_XXX, xxx_SensorInit, NULL },
#endif
#if defined(YYY_MIPI_RAW)
    { YYY_SENSOR_ID, SENSOR_DRVNAME_YYY, yyy_SensorInit, NULL },
#endif
    { 0, NULL, NULL }
};

sensor 驱动里的 Init + 函数表（xxx_sensor.c）：

C
static SENSOR_FUNCTION_STRUCT sensor_func = {
    .SensorOpen          = xxx_open,
    .SensorGetInfo       = xxx_get_info,
    .SensorGetResolution = xxx_get_resolution,
    .SensorFeatureControl= xxx_feature_control,
    .SensorControl       = xxx_control,
    .SensorClose         = xxx_close,
};

MSDK_SENSOR_RETURN_CODE_ENUM xxx_SensorInit(PSENSOR_FUNCTION_STRUCT *pfFunc)
{
    if (pfFunc != NULL)
        *pfFunc = &sensor_func;     // 把函数表交给调用者
    return ERROR_NONE;
}

kernel 侧 xxx_open 里实测 chip ID（配对真正的硬件依据）：

C
static kal_uint32 xxx_open(void)
{
    // 读 sensor 的 chip ID 寄存器 (地址随 sensor 而定, 如 0x0000/0x300A)
    readReg(XXX_CHIP_ID_REG, &chip_id);
    if (chip_id != XXX_SENSOR_ID)     // ★ 与列表里的 ID 比较
        return ERROR_SENSOR_CONNECT_FAIL;   // 不匹配 → 该 sensor 不在板
    return ERROR_NONE;
}

五、配对机制（关键）

graph TD A["xxx_sensor.h 中 #define XXX_SENSOR_ID 0x1234"] --> B["HAL sensorList[].sensorId = 0x1234"] A --> C["Kernel g_pstSensorList[].sensorId = 0x1234"] A --> D["Sensor 硬件 I2C chip ID = 0x1234"] B --> E["HAL 用 0x1234 找到函数表"] C --> F["Kernel 用 0x1234 找到驱动"] D --> F E --> G["两端通过 ID 对齐 → 配对成功"] F --> G
  • Init 函数名（HAL 的 xxx_SensorInit vs kernel 的 xxx_SensorInit）可以不同名，但 XXX_SENSOR_ID 必须完全一致。
  • 真正"有没有这个硬件"由 kernel xxx_open() 里的 I2C 读 ID 决定；读不到/不匹配 → 该槽位 sensor 视为不存在。

六、ioctl 通信链（HAL ↔ Kernel）

HAL 通过 /dev/kd_camera_hw 与 kernel 对话：

C
// HAL (ImgSensorDrv::init 简化)
fd = open("/dev/kd_camera_hw", O_RDWR);
ioctl(fd, KDIMGSENSORIOC_X_SET_DRIVER, &position);  // 指定前/后摄
ioctl(fd, KDIMGSENSORIOC_T_OPEN);                   // 触发 kernel 探测
// 之后 HAL 自己遍历 sensorList 取函数表
for (i=0; sensorList[i].sensorId; i++)
    sensorList[i].Init(&pFunc[i]);

Kernel 侧分发（kd_imgsensor.c）：

C
static long kdSensorIoctl(struct file *file, unsigned int cmd, unsigned long arg)
{
    switch (cmd) {
    case KDIMGSENSORIOC_T_OPEN:
        kdModuleInit();                 // 遍历 g_pstSensorList[], 调各 Init → I2C 探测
        break;
    case KDIMGSENSORIOC_X_SET_DRIVER:   // 设定当前 sensor
        g_sensorIdx = arg;
        break;
    case KDIMGSENSORIOC_X_FEATURECONCTROL:
        pSensorFunc->SensorFeatureControl(...);   // 转发到 sensor 驱动
        break;
    case KDIMGSENSORIOC_X_CONTROL:
        pSensorFunc->SensorControl(...);
        break;
    }
    return 0;
}

kdModuleInit() 探测逻辑：

C
for (i = 0; i < MAX_NUM_OF_SUPPORT_SENSOR; i++) {
    g_pstSensorList[i].Init(&g_pstSensorFunc[i]);  // 每个 sensor Init 内部读 I2C ID
    // 读到的 ID == sensorId 的, 标记为已连接
}

七、DTS 桥接（告诉 kernel 槽位信息）

// kernel-5.15/arch/arm64/boot/dts/mediatek/mt6893-<board>.dts
&kd_camera_hw1 {
    cam0_i2c_addr   = <0x20>;          // sensor 7-bit I2C 地址
    cam0_sensor_id  = <0x1234>;        // ★ 必须等于 XXX_SENSOR_ID
    cam0_sensor_name= "xxx_mipi_raw";
    cam0_mclk       = <0>;             // MCLK 通道
    cam0_rst_gpio   = <&pio 51 0>;
    cam0_pnd_gpio   = <&pio 52 0>;
    cam0_i2c_channel= <0>;             // i2c adapter
};

kernel kd_imgsensor probe 时解析这些属性，决定在哪个 I2C 总线上、用什么地址去读 cam0_sensor_id。

九、加一个新 sensor 的 Checklist

  1. xxx_sensor.h 定义 XXX_SENSOR_ID 和 SENSOR_DRVNAME_XXX（HAL/kernel 共用）。
  2. HAL：imgsensor_sensor_list.cpp 加一项 {ID, NAME, xxx_SensorInit}。
  3. Kernel：kd_imgsensor.c 的 g_pstSensorList[] 加同 ID 一项。
  4. Kernel：xxx_sensor.c 实现 xxx_SensorInit + SENSOR_FUNCTION_STRUCT（open 里读 chip ID 校验）。
  5. DTS：cam0_sensor_id = <XXX_SENSOR_ID> + I2C 地址/GPIO/MCLK。
  6. ProjectConfig.mk 开 XXX_MIPI_RAW = yes（宏控制 #if defined）。
  7. 编译： 
    mmm vendor/mediatek/proprietary/custom/mt6893/hal/imgsensor/
mmm vendor/mediatek/proprietary/hardware/mtkcam/
# kernel 侧整编或单独编 imgsensor
  8. 烧录：vendor_boot.img + dtbo.img + super.img（vendor）。
