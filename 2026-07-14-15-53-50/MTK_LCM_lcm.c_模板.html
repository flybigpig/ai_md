4. lcm.c（核心）

// lcm.c
#include <platform/mt_gpt.h>
#include <platform/disp_drv.h>
#include <platform/lcm_drv.h>
#include <platform/dsi_drv.h>
#include "lcm_drv.h"
#include "lcm_params.h"

/* ---------- 1. 屏厂 DSI 初始化序列 ---------- */
typedef struct {
    unsigned char cmd;
    unsigned char count;
    unsigned char para_list[64];
} LCM_setting_table;

// REGFLAG_END_OF_TABLE 习惯用 cmd=0xFF,count=0 当结束符（按你 BSP 约定）
static LCM_setting_table lcm_init_setting[] = {
    {0xFF, 3, {0x98, 0x81, 0x03}},   // 切页（示例，按屏厂）
    {0x11, 0, {}},                   // Exit Sleep
    {0x29, 0, {}},                   // Display On
    {0xFF, 0, {}},                   // END OF TABLE
};

static void push_table(LCM_setting_table *table, unsigned int cnt, unsigned char force)
{
    unsigned int i;
    for (i = 0; i < cnt; i++) {
        unsigned char cmd = table[i].cmd;
        if (cmd == 0xFF && table[i].count == 0) return;   // end
        if (cmd == 0xFE) { mdelay(table[i].count); continue; } // delay
        dsi_set_cmdq_pannel(cmd, table[i].count, table[i].para_list, force);
    }
}

/* ---------- 2. util funcs（基本空实现） ---------- */
static void lcm_set_util_funcs(const LCM_UTIL_FUNCS *util) { /* 通常留空 */ }

/* ---------- 3. get_params：填分辨率 + DSI 时序 + 时钟 ---------- */
static void lcm_get_params(LCM_PARAMS *params)
{
    memset(params, 0, sizeof(LCM_PARAMS));

    params->type   = LCM_TYPE_DSI;
    params->width  = LCM_WIDTH;
    params->height = LCM_HEIGHT;
    params->physical_width  = LCM_PHYSICAL_WIDTH_MM;
    params->physical_height = LCM_PHYSICAL_HEIGHT_MM;

    /* 数据格式 / 通道数 */
    params->dsi.mode                    = BURST_VDO_MODE; // 或 SYNC_PULSE_VDO_MODE
    params->dsi.data_format.color_order = LCM_COLOR_ORDER_RGB;
    params->dsi.data_format.trans_seq   = LCM_DSI_TRANS_SEQ_MSB_FIRST;
    params->dsi.data_format.format      = LCM_DSI_FORMAT_RGB888;
    params->dsi.lane_num                = LCM_4LANE;     // 屏厂接口：3/4 lane
    params->dsi.PS                      = LCM_PACKED_PS_24BIT_RGB888;
    params->dsi.intermediat_buffer_num  = 2;

    /* ---- 垂直时序（来自 datasheet 的 VFP/VBP/VSW） ---- */
    params->dsi.vertical_sync_active  = 4;    // VSW  vsync 脉宽
    params->dsi.vertical_back_porch   = 12;   // VBP
    params->dsi.vertical_front_porch  = 8;    // VFP
    params->dsi.vertical_active_line  = LCM_HEIGHT;

    /* ---- 水平时序（来自 datasheet 的 HFP/HBP/HSW） ---- */
    params->dsi.horizontal_sync_active  = 10; // HSW
    params->dsi.horizontal_back_porch   = 40; // HBP
    params->dsi.horizontal_front_porch  = 20; // HFP
    params->dsi.horizontal_active_pixel = LCM_WIDTH;

    /* ---- 时钟（DSI bit clock，单位约 MHz） ---- */
    params->dsi.ssc_disable = 0;   // 开 SSC 抗 EMI
    params->dsi.ssc_range   = 4;
    params->dsi.PLL_CLOCK   = 480; // 见下方"时钟怎么算"
}

/* ---------- 4. init / suspend / resume ---------- */
static void lcm_init(void)
{
    // 硬件 reset 由 LK platform 在调 init 前拉过；这里只发 DSI 序列
    push_table(lcm_init_setting,
               sizeof(lcm_init_setting)/sizeof(LCM_setting_table), 1);
}

static void lcm_suspend(void)
{
    unsigned char tmp = 0x00;
    dsi_set_cmdq_pannel(0x28, 0, &tmp, 1); // Display Off
    dsi_set_cmdq_pannel(0x10, 0, &tmp, 1); // Enter Sleep
}

static void lcm_resume(void)
{
    lcm_init();
}

static void lcm_set_backlight(unsigned int level)
{
    // level 0~255 → DCS 0x51 写背光
    unsigned char bl = (unsigned char)(level & 0xFF);
    dsi_set_cmdq_pannel(0x51, 1, &bl, 1);
}

/* ---------- 5. 注册 ---------- */
LCM_DRIVER xxx_1080p_lcm_drv = {
    .name          = LCM_DRIVER_NAME,
    .set_util_funcs= lcm_set_util_funcs,
    .get_params    = lcm_get_params,
    .init          = lcm_init,
    .suspend       = lcm_suspend,
    .resume        = lcm_resume,
    .set_backlight = lcm_set_backlight,
};

5. get_params 时序字段逐项解读

字段来自 datasheet填法
vertical_sync_activeVSW (Vsync width)datasheet 的 vsync pulse width，典型 2~10
vertical_back_porchVBPdatasheet 的 V back porch
vertical_front_porchVFPdatasheet 的 V front porch
horizontal_sync_activeHSWdatasheet 的 hsync pulse width
horizontal_back_porchHBPdatasheet 的 H back porch
horizontal_front_porchHFPdatasheet 的 H front porch
dsi.lane_num接口 lane 数屏厂给的 3-lane / 4-lane
dsi.PLL_CLOCKDSI bit clock见下方算法；决定了能否跑到目标帧率

PLL_CLOCK 怎么算

DSI 在每个 lane 上的数据率（bit rate）：

bit_rate  = (H_total × V_total × fps × bits_per_pixel) / lane_num
H_total   = horizontal_active + HFP + HBP + HSW
V_total   = vertical_active   + VFP + VBP + VSW

PLL_CLOCK ≈ bit_rate / 2      // LK 里 PLL_CLOCK 单位约为该半值(MHz)

例：1080×2400, 60fps, RGB888(24bpp), 4 lane
H_total = 1080+20+40+10 = 1150
V_total = 2400+8+12+4   = 2424
bit_rate= 1150×2424×60×24 / 4  ≈ 398 Mbps/lane
PLL_CLOCK ≈ 199 → 取向上取整的余量值（屏厂常给 ~480 是因为含 blanking 倍数/倍频约定，以屏厂给的为准）
PLL_CLOCK 填错最常见的现象：闪屏、竖纹、点不亮但 init 不报错。拿不准就直接用屏厂规格书里写的 DSI clock。

7. LK 与 kernel 两份的差异

点LK 侧Kernel 侧
路径bootable/bootloader/lk/dev/lcm/kernel-5.15/drivers/misc/mediatek/lcm/
头文件platform/lcm_drv.hmtk_lcm_drv.h 等内核头
DSI 发命令dsi_set_cmdq_pannel()走 DRM/DSI 子系统的 mipi_dsi_dcs_write()
probe无，列表枚举有 probe()，DTS 匹配 lcm_name
名字一致性两份的 panel name + DTS lcm_name 必须完全一致，否则 kernel 起屏黑
屏不亮排查顺序（实战）：
  1. LK 和 kernel 的 LCM 名字是否一致？
  2. reset / power GPIO 与硬件原理图对得上？（DTS reset-gpios / power-gpios）
  3. DSI 初始化序列是否完整（页切换、Exit Sleep、Display On 顺序）？
  4. PLL_CLOCK 是否按屏厂值？
  5. kernel log 搜 lcm / panel 看 probe 是否成功。
